#End If

' Convert UserForm points to screen pixels (DPI aware)
' -------------------------------------------------------------
#If VBA7 Then
Public Function PtsToPixels(pts As Single, bVert As Boolean) As Long
    Dim hDC As LongPtr
#Else
Public Function PtsToPixels(pts As Single, bVert As Boolean) As Long
    Dim hDC As Long
#End If
    hDC = GetDC(0)
    If bVert Then
        PtsToPixels = CLng((pts / 72) * GetDeviceCaps(hDC, 90))
    Else
        PtsToPixels = CLng((pts / 72) * GetDeviceCaps(hDC, 88))
    End If
    ReleaseDC 0, hDC
End Function

' Remove the title bar and apply rounded corners
' -------------------------------------------------------------
Public Sub ApplyRoundForm(frm As Object, Optional radius As Long = 25)
#If VBA7 Then
    Dim hWnd As LongPtr, hRgn As LongPtr
#Else
    Dim hWnd As Long, hRgn As Long
#End If
    If frm.Caption = "" Then
        hWnd = FindWindow("ThunderDFrame", vbNullString)
    Else
        hWnd = FindWindow("ThunderDFrame", frm.Caption)
    End If
    If hWnd = 0 Then Exit Sub
    
    ' 1. Remove the title bar style (WS_CAPTION) and modal border (WS_EX_DLGMODALFRAME)
    SetWindowLong hWnd, -16, GetWindowLong(hWnd, -16) And Not &HC00000
    SetWindowLong hWnd, -20, GetWindowLong(hWnd, -20) And Not &H1
    
    ' 2. Create a rounded rectangle region and apply it to the form
    hRgn = CreateRoundRectRgn(0, 0, PtsToPixels(frm.Width, False), PtsToPixels(frm.Height, True), radius, radius)
    SetWindowRgn hWnd, hRgn, True
    DrawMenuBar hWnd
End Sub
